-- Set the lua name
api:set_lua_name("Snowball Manager")

-- // Services
local players_service = cloneref(game:GetService("Players"))
local local_player = players_service.LocalPlayer
local replicated_storage = cloneref(game:GetService("ReplicatedStorage"))
local run_service = cloneref(game:GetService("RunService"))

-- // Config
local snowball_location = CFrame.new(0, 10, 0) -- Default location
local is_grabbing = false

----------------------------------------------------------------------
-- Desync Callback
----------------------------------------------------------------------

api:add_desync_callback(2, function()
    if is_grabbing then
        return snowball_location
    end
    return nil
end)

----------------------------------------------------------------------
-- Snowball Functions
----------------------------------------------------------------------

local function grab_snowballs()
    task.spawn(function()
        if not api:can_desync() then
            api:notify("⚠️ Cannot desync right now!", 2)
            return
        end
        
        local amount = Options.snowball_amount and Options.snowball_amount.Value or 1
        local delay = Options.grab_delay and Options.grab_delay.Value or 1
        
        api:notify("🔄 Desyncing to snowball location...", 2)
        is_grabbing = true
        task.wait(0.3) -- Wait for desync to activate
        
        api:notify("⛄ Grabbing " .. amount .. " snowball(s) (Delay: " .. delay .. "s)...", 2)
        
        for i = 1, amount do
            pcall(function()
                local args = {"PickSnow"}
                replicated_storage:WaitForChild("MainEvent"):FireServer(unpack(args))
            end)
            
            if i < amount then
                api:notify("⛄ Grabbed " .. i .. "/" .. amount .. " snowballs...", 1)
                task.wait(delay) -- Use configurable delay
            end
        end
        
        is_grabbing = false -- Stop desyncing
        api:notify("✅ Finished grabbing " .. amount .. " snowballs!", 2)
    end)
end

local function throw_all_snowballs()
    task.spawn(function()
        local count = 0
        local throw_delay = Options.throw_delay and Options.throw_delay.Value or 0.2
        
        -- Count snowballs in backpack
        for _, tool in ipairs(local_player.Backpack:GetChildren()) do
            if tool.Name:lower():find("snow") or (tool.ToolTip and tool.ToolTip:lower():find("snow")) then
                count += 1
            end
        end
        
        if count == 0 then
            api:notify("❌ No snowballs in backpack!", 2)
            return
        end
        
        api:notify("❄️ Throwing " .. count .. " snowballs (Delay: " .. throw_delay .. "s)...", 2)
        
        local thrown = 0
        
        -- Equip and throw each snowball
        for _, tool in ipairs(local_player.Backpack:GetChildren()) do
            if tool.Name:lower():find("snow") or (tool.ToolTip and tool.ToolTip:lower():find("snow")) then
                tool.Parent = local_player.Character
                task.wait(0.05)
                tool:Activate()
                task.wait(0.05)
                tool:Activate()
                
                thrown += 1
                
                -- Wait between each snowball throw (except for the last one)
                if thrown < count then
                    api:notify("⛄ Threw " .. thrown .. "/" .. count .. " snowballs...", 1)
                    task.wait(throw_delay)
                end
            end
        end
        
        api:notify("⛄ ALL " .. count .. " SNOWBALLS THROWN! ⛄", 3)
    end)
end

local function set_snowball_location()
    if local_player.Character and local_player.Character:FindFirstChild("HumanoidRootPart") then
        snowball_location = local_player.Character.HumanoidRootPart.CFrame
        api:notify("📍 Snowball location set to current position!", 2)
    else
        api:notify("❌ Character not found!", 2)
    end
end

local function auto_farm()
    task.spawn(function()
        local auto_enabled = Toggles.auto_farm_toggle and Toggles.auto_farm_toggle.Value or false
        
        while auto_enabled and task.wait(1) do
            auto_enabled = Toggles.auto_farm_toggle and Toggles.auto_farm_toggle.Value or false
            
            if not auto_enabled then break end
            
            -- Count current snowballs
            local current_count = 0
            for _, tool in ipairs(local_player.Backpack:GetChildren()) do
                if tool.Name:lower():find("snow") or (tool.ToolTip and tool.ToolTip:lower():find("snow")) then
                    current_count += 1
                end
            end
            
            local max_amount = Options.auto_farm_max and Options.auto_farm_max.Value or 10
            
            if current_count < max_amount then
                grab_snowballs()
                local grab_delay = Options.grab_delay and Options.grab_delay.Value or 1
                local snowball_amount = Options.snowball_amount and Options.snowball_amount.Value or 1
                -- Wait for all snowballs to be grabbed
                task.wait((snowball_amount * grab_delay) + 2)
            end
            
            task.wait(2)
        end
    end)
end
