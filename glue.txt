-- Function to enable glue
local function EnableGlue(target)
    if not target or not target.Character or not target.Character:FindFirstChild("HumanoidRootPart") then
        return false
    end
    
    local TargetPart = target.Character.HumanoidRootPart
    
    if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
        -- Set physics attachment
        pcall(function()
            sethiddenproperty(LocalPlayer.Character.HumanoidRootPart, "PhysicsRepRootPart", TargetPart)
        end)
        
        -- Calculate base desync CFrame
        local desyncCFrame = CFrame.new(TargetPart.Position + GlueOffset) * GlueAngle
        
        -- Apply "Always Face Target" rotation if enabled
        if Toggles['always_face_target'] and Toggles['always_face_target'].Value then
            local myRoot = LocalPlayer.Character.HumanoidRootPart
            local myPos = myRoot.Position
            local targetPos = TargetPart.Position
            local faceMode = Options['face_mode'].Value
            
            local lookDirection
            if faceMode == "Face Target" then
                -- Look directly at target
                lookDirection = (targetPos - myPos).Unit
            elseif faceMode == "Behind Target" then
                -- Look in same direction as target (behind them)
                lookDirection = TargetPart.CFrame.LookVector
            elseif faceMode == "Side View Left" then
                -- Look from left side
                lookDirection = TargetPart.CFrame.RightVector
            elseif faceMode == "Side View Right" then
                -- Look from right side
                lookDirection = -TargetPart.CFrame.RightVector
            end
            
            -- Create rotation CFrame to face the direction
            if lookDirection then
                local facingCFrame = CFrame.lookAt(desyncCFrame.Position, desyncCFrame.Position + lookDirection)
                -- Preserve the pitch/roll from GlueAngle but apply new facing direction
                desyncCFrame = facingCFrame * CFrame.Angles(GlueAngle:ToEulerAnglesYXZ())
            end
        end
        
        api:set_desync_cframe(desyncCFrame)
        return true
    end
    return false
end

-- Function to disable glue
local function DisableGlue()
    if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
        pcall(function()
            sethiddenproperty(LocalPlayer.Character.HumanoidRootPart, "PhysicsRepRootPart", nil)
        end)
    end
end

-- Main glue loop
local CurrentTarget = nil
Handler:AddConnection("Sticky Glue Update", RunService.Heartbeat:Connect(function()
    if not Toggles['glue_enabled'] or not Toggles['glue_enabled'].Value or not Options['glue_keybind']:GetState() then
        if CurrentTarget then
            DisableGlue()
            CurrentTarget = nil
        end
        return
    end
    
    -- Get target based on selected type
    local targetType = Options['glue_target_type'].Value:lower()
    local Target = api:get_target(targetType)
    
    -- Check if we should only glue when target exists
    if Toggles['only_when_target'].Value and not Target then
        if CurrentTarget then
            DisableGlue()
            CurrentTarget = nil
            if Toggles['show_notifications'].Value then
                api:notify("Target lost - unglued", 2)
            end
        end
        return
    end
    
    -- If we have a target, glue to them
    if Target then
        -- Notify when we get a new target
        if CurrentTarget ~= Target then
            CurrentTarget = Target
            if Toggles['show_notifications'].Value then
                api:notify("Glued to " .. Target.Name, 2)
            end
        end
        
        -- Update glue position every frame if enabled
        if Toggles['auto_update_position'].Value then
            EnableGlue(Target)
        end
    end
end))

-- Unload handler
api:on_event("unload", function()
    DisableGlue()
    api:notify("Sticky glue unloaded", 2)
end)

api:notify("Sticky glue loaded - Press G to toggle", 3)
