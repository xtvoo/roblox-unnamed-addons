-- Levenshtein Distance for spell-check/auto-correct
local function LevenshteinDistance(s1, s2)
    local len1, len2 = #s1, #s2
    
    if len1 == 0 then return len2 end
    if len2 == 0 then return len1 end
    
    local matrix = {}
    for i = 0, len1 do
        matrix[i] = {[0] = i}
    end
    for j = 0, len2 do
        matrix[0][j] = j
    end
    
    for i = 1, len1 do
        for j = 1, len2 do
            local cost = (s1:sub(i, i) == s2:sub(j, j)) and 0 or 1
            matrix[i][j] = math.min(
                matrix[i-1][j] + 1,      -- deletion
                matrix[i][j-1] + 1,      -- insertion
                matrix[i-1][j-1] + cost  -- substitution
            )
        end
    end
    
    return matrix[len1][len2]
end

-- Auto-populate skin database from game
local SKIN_DATABASE = {}
local CATEGORY_LIST = {}

local function LoadSkinsFromGame()
    local skinAssets = ReplicatedStorage:FindFirstChild("SkinAssets")
    if not skinAssets then
        print("WARNING: SkinAssets not found, using hardcoded database")
        return false
    end
    
    local skinScriptsStorage = skinAssets:FindFirstChild("SkinScriptsStorage")
    if not skinScriptsStorage then
        print("WARNING: SkinScriptsStorage not found, using hardcoded database")
        return false
    end
    
    print("=== LOADING SKINS FROM GAME ===")
    
    for _, categoryFolder in pairs(skinScriptsStorage:GetChildren()) do
        if categoryFolder:IsA("Folder") then
            local categoryName = categoryFolder.Name
            
            if string.sub(categoryName, 1, 1) == "[" then
                SKIN_DATABASE[categoryName] = {}
                table.insert(CATEGORY_LIST, categoryName)
                
                print("Category: " .. categoryName)
                
                for _, skinItem in pairs(categoryFolder:GetChildren()) do
                    local skinName = skinItem.Name
                    table.insert(SKIN_DATABASE[categoryName], skinName)
                end
                
                table.sort(SKIN_DATABASE[categoryName])
            end
        end
    end
    
    table.sort(CATEGORY_LIST)
    
    print("=== LOADED " .. #CATEGORY_LIST .. " CATEGORIES FROM GAME ===")
    return #CATEGORY_LIST > 0
end

        table.insert(CATEGORY_LIST, category)
    end
    table.sort(CATEGORY_LIST)

end

-- Helper function to parse and check skins
local function PlayerHasSkin(player, category, skinName)
    local dataFolder = player:FindFirstChild("DataFolder")
    if not dataFolder then return false, {}, 0 end
    
    local results = {
        EquipSkins = 0,
        WeaponSkins = 0,
        Skins = 0
    }
    
    local totalQuantity = 0
    
    local skinsValue = dataFolder:FindFirstChild("Skins")
    if skinsValue and skinsValue:IsA("ValueBase") then
        local success, skinData = pcall(function()
            return HttpService:JSONDecode(skinsValue.Value)
        end)
        
        if success and skinData then
            if skinData[category] and skinData[category][skinName] then
                local quantity = skinData[category][skinName]
                if quantity >= 1 then
                    results.Skins = quantity
                    totalQuantity = totalQuantity + quantity
                end
            end
        end
    end
    
    local information = dataFolder:FindFirstChild("Information")
    if information then
        local equipSkins = information:FindFirstChild("EquipSkins")
        if equipSkins and equipSkins:IsA("ValueBase") then
            local success, equipData = pcall(function()
                return HttpService:JSONDecode(equipSkins.Value)
            end)
            
            if success and equipData then
                if equipData[category] and equipData[category][skinName] then
                    local quantity = equipData[category][skinName]
                    if quantity >= 1 then
                        results.EquipSkins = quantity
                    end
                end
            end
        end
        
        local weaponSkins = information:FindFirstChild("WeaponSkins")
        if weaponSkins and weaponSkins:IsA("ValueBase") then
            local success, weaponData = pcall(function()
                return HttpService:JSONDecode(weaponSkins.Value)
            end)
            
            if success and weaponData then
                if weaponData[category] and weaponData[category][skinName] then
                    local quantity = weaponData[category][skinName]
                    if quantity >= 1 then
                        results.WeaponSkins = quantity
                    end
                end
            end
        end
    end
    
    local hasAnySkin = results.EquipSkins > 0 or results.WeaponSkins > 0 or results.Skins > 0
    return hasAnySkin, results, totalQuantity
end

-- Search/Auto-correct input
MainBox:AddInput("skin_search_input", {
    Default = "",
    Numeric = false,
    Finished = false,
    Text = "Or type to search",
    Tooltip = "Type a skin name and it will auto-correct",
    Placeholder = "Type 'gpo' or 'battleworn'...",
    Callback = function(value)
        if value == "" or value == " " then return end
        
        local searchTerm = value:lower()
        local availableSkins = SKIN_DATABASE[SELECTED_CATEGORY] or {}
        local closestMatch = nil
        local closestDistance = math.huge
        
        -- First pass: exact match or contains substring
        for _, skinName in ipairs(availableSkins) do
            local skinLower = skinName:lower()
            
            -- Exact match (best)
            if skinLower == searchTerm then
                closestMatch = skinName
                closestDistance = 0
                break
            end
            
            -- Contains substring
            if string.find(skinLower, searchTerm, 1, true) then
                local distance = math.abs(#skinLower - #searchTerm)
                if distance < closestDistance then
                    closestMatch = skinName
                    closestDistance = distance
                end
            end
        end
        
        -- Second pass: Levenshtein distance if no substring match
        if not closestMatch or closestDistance > 0 then
            for _, skinName in ipairs(availableSkins) do
                local skinLower = skinName:lower()
                local distance = LevenshteinDistance(searchTerm, skinLower)
                
                if distance < closestDistance then
                    closestMatch = skinName
                    closestDistance = distance
                end
            end
        end
        
        if closestMatch then
            SELECTED_SKIN = closestMatch
            api:notify("Auto-corrected to: " .. closestMatch, 2)
        else
            api:notify("No skin found matching: " .. value, 3)
        end
    end
})
