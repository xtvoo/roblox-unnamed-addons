-- Levenshtein Distance for spell-check/auto-correct
local function LevenshteinDistance(s1, s2)
    local len1, len2 = #s1, #s2
    
    if len1 == 0 then return len2 end
    if len2 == 0 then return len1 end
    
    local matrix = {}
    for i = 0, len1 do
        matrix[i] = {[0] = i}
    end
    for j = 0, len2 do
        matrix[0][j] = j
    end
    
    for i = 1, len1 do
        for j = 1, len2 do
            local cost = (s1:sub(i, i) == s2:sub(j, j)) and 0 or 1
            matrix[i][j] = math.min(
                matrix[i-1][j] + 1,      -- deletion
                matrix[i][j-1] + 1,      -- insertion
                matrix[i-1][j-1] + cost  -- substitution
            )
        end
    end
    
    return matrix[len1][len2]
end

-- Auto-populate skin database from game
local SKIN_DATABASE = {}
local CATEGORY_LIST = {}

local function LoadSkinsFromGame()
    local skinAssets = ReplicatedStorage:FindFirstChild("SkinAssets")
    if not skinAssets then
        print("WARNING: SkinAssets not found, using hardcoded database")
        return false
    end
    
    local skinScriptsStorage = skinAssets:FindFirstChild("SkinScriptsStorage")
    if not skinScriptsStorage then
        print("WARNING: SkinScriptsStorage not found, using hardcoded database")
        return false
    end
    
    print("=== LOADING SKINS FROM GAME ===")
    
    for _, categoryFolder in pairs(skinScriptsStorage:GetChildren()) do
        if categoryFolder:IsA("Folder") then
            local categoryName = categoryFolder.Name
            
            if string.sub(categoryName, 1, 1) == "[" then
                SKIN_DATABASE[categoryName] = {}
                table.insert(CATEGORY_LIST, categoryName)
                
                print("Category: " .. categoryName)
                
                for _, skinItem in pairs(categoryFolder:GetChildren()) do
                    local skinName = skinItem.Name
                    table.insert(SKIN_DATABASE[categoryName], skinName)
                end
                
                table.sort(SKIN_DATABASE[categoryName])
            end
        end
    end
    
    table.sort(CATEGORY_LIST)
    
    print("=== LOADED " .. #CATEGORY_LIST .. " CATEGORIES FROM GAME ===")
    return #CATEGORY_LIST > 0
end

-- Try to load from game first

        table.insert(CATEGORY_LIST, category)
    end
    table.sort(CATEGORY_LIST)
    api:notify("Using hardcoded skins database", 2)
end

-- Helper function to parse and check skins
local function PlayerHasSkin(player, category, skinName)
    local dataFolder = player:FindFirstChild("DataFolder")
    if not dataFolder then return false, {}, 0 end
    
    local results = {
        EquipSkins = 0,
        WeaponSkins = 0,
        Skins = 0
    }
    
    local totalQuantity = 0
    
    local skinsValue = dataFolder:FindFirstChild("Skins")
    if skinsValue and skinsValue:IsA("ValueBase") then
        local success, skinData = pcall(function()
            return HttpService:JSONDecode(skinsValue.Value)
        end)
        
        if success and skinData then
            if skinData[category] and skinData[category][skinName] then
                local quantity = skinData[category][skinName]
                if quantity >= 1 then
                    results.Skins = quantity
                    totalQuantity = totalQuantity + quantity
                end
            end
        end
    end
    
    local information = dataFolder:FindFirstChild("Information")
    if information then
        local equipSkins = information:FindFirstChild("EquipSkins")
        if equipSkins and equipSkins:IsA("ValueBase") then
            local success, equipData = pcall(function()
                return HttpService:JSONDecode(equipSkins.Value)
            end)
            
            if success and equipData then
                if equipData[category] and equipData[category][skinName] then
                    local quantity = equipData[category][skinName]
                    if quantity >= 1 then
                        results.EquipSkins = quantity
                    end
                end
            end
        end
        
        local weaponSkins = information:FindFirstChild("WeaponSkins")
        if weaponSkins and weaponSkins:IsA("ValueBase") then
            local success, weaponData = pcall(function()
                return HttpService:JSONDecode(weaponSkins.Value)
            end)
            
            if success and weaponData then
                if weaponData[category] and weaponData[category][skinName] then
                    local quantity = weaponData[category][skinName]
                    if quantity >= 1 then
                        results.WeaponSkins = quantity
                    end
                end
            end
        end
    end
    
    local hasAnySkin = results.EquipSkins > 0 or results.WeaponSkins > 0 or results.Skins > 0
    return hasAnySkin, results, totalQuantity
end

MainBox:AddButton({
    Text = "Search Skin",
    Func = function()
        local searchInput = Options.skin_search_input.Value
        
        if searchInput == "" or searchInput == " " then
            api:notify("Please type a skin name!", 2)
            return
        end
        
        -- Split search input into multiple words
        local searchWords = {}
        for word in searchInput:gmatch("%S+") do
            table.insert(searchWords, word:lower())
        end
        
        local availableSkins = SKIN_DATABASE[SELECTED_CATEGORY] or {}
        local closestMatch = nil
        local bestScore = 0
        local matchDetails = ""
        
        print("\n=== SEARCHING FOR: " .. searchInput .. " ===")
        print("Category: " .. SELECTED_CATEGORY)
        print("Search words: " .. table.concat(searchWords, ", "))
        print("Total skins available: " .. #availableSkins)
        
        -- Helper: check if all search words are in skin name
        local function allWordsMatch(skinName)
            local skinLower = skinName:lower()
            local matchCount = 0
            
            for _, searchWord in ipairs(searchWords) do
                if string.find(skinLower, searchWord, 1, true) then
                    matchCount = matchCount + 1
                end
            end
            
            return matchCount == #searchWords
        end
        
        -- Helper: count how many words match
        local function countMatches(skinName)
            local skinLower = skinName:lower()
            local matchCount = 0
            
            for _, searchWord in ipairs(searchWords) do
                if string.find(skinLower, searchWord, 1, true) then
                    matchCount = matchCount + 1
                end
            end
            
            return matchCount
        end
        
        -- Strategy 1: ALL WORDS MUST MATCH + EXACT STARTS
        for _, skinName in ipairs(availableSkins) do
            local skinLower = skinName:lower()
            
            -- Check if skin name starts with first search word
            if string.sub(skinLower, 1, #searchWords[1]) == searchWords[1] and allWordsMatch(skinName) then
                closestMatch = skinName
                bestScore = 1000
                matchDetails = "ALL WORDS + STARTS WITH"
                print("✓ " .. matchDetails .. ": " .. skinName)
                break
            end
        end
        
        -- Strategy 2: ALL WORDS MATCH (in any order)
        if not closestMatch then
            for _, skinName in ipairs(availableSkins) do
                if allWordsMatch(skinName) then
                    closestMatch = skinName
                    bestScore = 900
                    matchDetails = "ALL WORDS MATCH"
                    print("✓ " .. matchDetails .. ": " .. skinName)
                    break
                end
            end
        end
        
        -- Strategy 3: MOST WORDS MATCH
        if not closestMatch then
            local bestMatchCount = 0
            
            for _, skinName in ipairs(availableSkins) do
                local matchCount = countMatches(skinName)
                
                if matchCount > bestMatchCount then
                    bestMatchCount = matchCount
                    closestMatch = skinName
                    bestScore = 500 + (bestMatchCount * 100)
                    matchDetails = bestMatchCount .. "/" .. #searchWords .. " words match"
                    print("✓ " .. matchDetails .. ": " .. skinName)
                end
            end
        end
        
        -- Strategy 4: SINGLE WORD MATCH (if only 1 search word)
        if not closestMatch and #searchWords == 1 then
            local searchTerm = searchWords[1]
            
            for _, skinName in ipairs(availableSkins) do
                local skinLower = skinName:lower()
                
                if string.sub(skinLower, 1, #searchTerm) == searchTerm then
                    closestMatch = skinName
                    bestScore = 300
                    matchDetails = "STARTS WITH"
                    print("✓ " .. matchDetails .. ": " .. skinName)
                    break
                end
            end
            
            if not closestMatch then
                for _, skinName in ipairs(availableSkins) do
                    local skinLower = skinName:lower()
                    if string.find(skinLower, searchTerm, 1, true) then
                        closestMatch = skinName
                        bestScore = 200
                        matchDetails = "CONTAINS"
                        print("✓ " .. matchDetails .. ": " .. skinName)
                        break
                    end
                end
            end
        end
        
        if closestMatch then
            SELECTED_SKIN = closestMatch
            
            -- Also update the dropdown
            local skinDropdown = api:get_ui_object("skin_selection")
            if skinDropdown then
                skinDropdown:SetValue(closestMatch)
            end
            
            api:notify("✓ Found: " .. closestMatch .. " (" .. matchDetails .. ")", 2)
            print("SELECTED: " .. closestMatch .. " (score: " .. bestScore .. ", " .. matchDetails .. ")\n")
        else
            api:notify("✗ No match found for: " .. searchInput, 3)
            print("NO MATCH FOUND for: " .. searchInput .. "\n")
            
            -- Show available skins in this category
            print("Available skins in " .. SELECTED_CATEGORY .. ":")
            for i, skin in ipairs(availableSkins) do
                if i <= 15 then
                    print("  " .. i .. ". " .. skin)
                end
            end
            if #availableSkins > 15 then
                print("  ... and " .. (#availableSkins - 15) .. " more")
            end
        end
    end
})

MainBox:AddButton({
    Text = "Auto-Correct Target Name",
    Func = function()
        local inputName = Options.target_player_input.Value
        
        if inputName == "" or inputName == " " then
            api:notify("Please enter a player name!", 2)
            return
        end
        
        local inputLower = inputName:lower()
        local closestMatch = nil
        local closestDistance = math.huge
        local allPlayers = Players:GetPlayers()
        
        print("\n=== SEARCHING FOR PLAYER: " .. inputName .. " ===")
        print("Total players: " .. #allPlayers)
        
        -- Exact match first
        for _, player in pairs(allPlayers) do
            if player.Name:lower() == inputLower then
                closestMatch = player.Name
                closestDistance = 0
                print("✓ EXACT MATCH: " .. player.Name)
                break
            end
        end
        
        -- Starts with
        if not closestMatch then
            for _, player in pairs(allPlayers) do
                local playerLower = player.Name:lower()
                if string.sub(playerLower, 1, #inputLower) == inputLower then
                    closestMatch = player.Name
                    closestDistance = 0
                    print("✓ STARTS WITH: " .. player.Name)
                    break
                end
            end
        end
        
        -- Contains substring
        if not closestMatch then
            for _, player in pairs(allPlayers) do
                local playerLower = player.Name:lower()
                if string.find(playerLower, inputLower, 1, true) then
                    closestMatch = player.Name
                    closestDistance = math.abs(#playerLower - #inputLower)
                    print("✓ CONTAINS: " .. player.Name)
                    break
                end
            end
        end
        
        -- Levenshtein distance
        if not closestMatch then
            for _, player in pairs(allPlayers) do
                local distance = LevenshteinDistance(inputLower, player.Name:lower())
                if distance < closestDistance then
                    closestMatch = player.Name
                    closestDistance = distance
                    print("✓ FUZZY MATCH: " .. player.Name .. " (distance: " .. distance .. ")")
                end
            end
        end
        
        if closestMatch and closestDistance <= 3 then
            Options.target_player_input:SetValue(closestMatch)
            api:notify("✓ Auto-corrected to: " .. closestMatch, 2)
            print("SELECTED TARGET: " .. closestMatch .. "\n")
        else
            api:notify("✗ No player found matching: " .. inputName, 3)
            print("NO MATCH FOUND\n")
        end
    end
})
