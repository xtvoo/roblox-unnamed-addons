-- // Configuration
local Config = {
    Beam = {
        Enabled = true,
        StartColor = Color3.fromRGB(255, 220, 100),
        EndColor = Color3.fromRGB(255, 80, 80),
        Width = 0.12,
        Lifetime = 0.35,
        Brightness = 2.2,
        FadeLerp = 0.7,
        PositionLerp = 0.0
    },
    Impact = {
        Enabled = true,
        Color = Color3.fromRGB(255, 60, 60),
        Size = 3.6,
        Shape = "Sphere", 
        Lifetime = 0.65,
        FadeLerp = 0.55
    }
}

-- // Utilities
local function sanitizeColor(c)
    if c.R > 1 or c.G > 1 or c.B > 1 then
        return Color3.new(c.R/255, c.G/255, c.B/255)
    end
    return c
end

-- // Effect Logic
local Effects = {}

function Effects.SpawnBeam(from, to)
    if not Config.Beam.Enabled then return end
    
    local dist = (to - from).Magnitude
    if dist < 0.1 then return end

    local beamPart = Instance.new("Part")
    beamPart.Name = "FX_Beam"
    beamPart.Anchored = true
    beamPart.CanCollide = false
    beamPart.Transparency = 1
    beamPart.Size = Vector3.new(0.05, 0.05, 0.05)
    beamPart.Position = from
    beamPart.Parent = workspace

    local att0 = Instance.new("Attachment", beamPart)
    local att1 = Instance.new("Attachment", beamPart)
    
    local beam = Instance.new("Beam")
    beam.Attachment0 = att0
    beam.Attachment1 = att1
    beam.Width0 = Config.Beam.Width
    beam.Width1 = Config.Beam.Width
    beam.FaceCamera = true
    beam.LightEmission = Config.Beam.Brightness
    
    local sColor = sanitizeColor(Config.Beam.StartColor)
    local eColor = sanitizeColor(Config.Beam.EndColor)
    beam.Color = ColorSequence.new(sColor, eColor)
    
    beam.Transparency = NumberSequence.new(0)
    beam.Parent = beamPart

    local lifetime = Config.Beam.Lifetime
    local travelLerp = math.clamp(Config.Beam.PositionLerp, 0, 1)
    
    local travelTime = 0
    if travelLerp > 0 then
        att1.Position = Vector3.new(0,0,0)
        travelTime = lifetime * 0.85 * travelLerp
        
        local tweenMoves = game:GetService("TweenService"):Create(att1, TweenInfo.new(travelTime, Enum.EasingStyle.Linear), {
            Position = beamPart.CFrame:PointToObjectSpace(to)
        })
        tweenMoves:Play()
    else
        att1.WorldPosition = to
    end

    local fadeTime = lifetime * math.clamp(Config.Beam.FadeLerp, 0, 1)
    local delayTime = lifetime - fadeTime
    
    -- Cleanup Sequence
    task.delay(delayTime, function()
        if not beam or not beam.Parent then return end
        local tweenFade = game:GetService("TweenService"):Create(beam, TweenInfo.new(fadeTime, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
            Transparency = NumberSequence.new(1)
        })
        tweenFade:Play()
    end)
    
    task.delay(lifetime + 0.1, function()
        if beamPart and beamPart.Parent then beamPart:Destroy() end
    end)
end

function Effects.SpawnImpact(pos)
    if not Config.Impact.Enabled then return end

    local p = Instance.new("Part")
    p.Name = "FX_Impact"
    p.Anchored = true
    p.CanCollide = false
    p.Material = Enum.Material.Neon
    p.Color = sanitizeColor(Config.Impact.Color)
    p.CFrame = CFrame.new(pos)
    p.Transparency = 0
    p.Parent = workspace

    local rawSize = Config.Impact.Size / 10
    if Config.Impact.Shape == "Sphere" then
        p.Shape = Enum.PartType.Ball
        p.Size = Vector3.new(rawSize, rawSize, rawSize)
    else
        p.Shape = Enum.PartType.Block
        p.Size = Vector3.new(rawSize * 2, rawSize, rawSize * 3)
    end

    local lifetime = Config.Impact.Lifetime
    local fadeTime = lifetime * math.clamp(Config.Impact.FadeLerp, 0, 1)
    local delayTime = lifetime - fadeTime

    task.delay(delayTime, function()
        if not p or not p.Parent then return end
        local tween = game:GetService("TweenService"):Create(p, TweenInfo.new(fadeTime, Enum.EasingStyle.Linear), {
            Transparency = 1
        })
        tween:Play()
    end)
    
    task.delay(lifetime + 0.1, function()
        if p and p.Parent then p:Destroy() end
    end)
end
