-- Levenshtein distance for fuzzy matching
local function levenshtein(str1, str2)
    local len1, len2 = #str1, #str2
    local matrix = {}
    
    for i = 0, len1 do
        matrix[i] = {[0] = i}
    end
    
    for j = 0, len2 do
        matrix[0][j] = j
    end
    
    for i = 1, len1 do
        for j = 1, len2 do
            local cost = (str1:sub(i, i) == str2:sub(j, j)) and 0 or 1
            matrix[i][j] = math.min(
                matrix[i-1][j] + 1,
                matrix[i][j-1] + 1,
                matrix[i-1][j-1] + cost
            )
        end
    end
    
    return matrix[len1][len2]
end

-- Find player by partial name with fuzzy matching
local function Get_Player(name)
    if not name or name == "" then
        return nil
    end
    
    local players = Players:GetPlayers()
    local bestMatch = nil
    local bestScore = math.huge
    local lowerName = string.lower(name)
    
    for _, player in ipairs(players) do
        local userName = string.lower(player.Name)
        local displayName = string.lower(player.DisplayName)
        
        if userName:find(lowerName, 1, true) == 1 then
            return player
        end
        
        if displayName:find(lowerName, 1, true) == 1 then
            return player
        end
        
        if userName:find(lowerName, 1, true) or displayName:find(lowerName, 1, true) then
            local matchLength = math.max(#lowerName, 1)
            if matchLength > bestScore or bestScore == math.huge then
                bestMatch = player
                bestScore = matchLength
            end
        end
        
        local userDistance = levenshtein(lowerName, userName)
        local displayDistance = levenshtein(lowerName, displayName)
        local minDistance = math.min(userDistance, displayDistance)
        
        if minDistance < bestScore and minDistance <= 3 then
            bestMatch = player
            bestScore = minDistance
        end
    end
    
    return bestMatch
end

-- Parse command arguments
local function ParseArgs(message, prefix)
    local args = {}
    local command = message:match("^" .. prefix .. "(%S+)")
    local argString = message:match("^" .. prefix .. "%S+%s+(.+)") or ""
    
    for arg in argString:gmatch("[^,%s]+") do
        table.insert(args, arg)
    end
    
    return command, args, argString
end

            table.insert(normalWhitelist, owner.Name)
            SetOption("combat_whitelist", normalWhitelist)

-- Check if player is knocked
local function IsKO(player)
    if player and player.Character then
        local bodyEffects = player.Character:FindFirstChild("BodyEffects")
        if bodyEffects and bodyEffects:FindFirstChild("K.O") then
            return bodyEffects["K.O"].Value
        end
    end
    return false
end

-- Check if player is dead
local function IsDead(player)
    if player and player.Character then
        local bodyEffects = player.Character:FindFirstChild("BodyEffects")
        if bodyEffects and bodyEffects:FindFirstChild("Dead") then
            return bodyEffects["Dead"].Value
        end
    end
    return false
end

-- Find who is carrying a given target by scanning all characters
local function GetCarrier(targetPlayer)
    if not targetPlayer or not targetPlayer.Character then
        return nil
    end

    local targetChar = targetPlayer.Character

    -- Loop every player except the target and check their character for a grab constraint
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= targetPlayer and plr.Character then
            -- Try to find any grabbing constraint inside their character
            local grabConstraint = plr.Character:FindFirstChild("GRABBING_CONSTRAINT", true)
            if grabConstraint then
                -- Most grabs weld/attach the target's HumanoidRootPart/UpperTorso/etc.
                local attachment0 = grabConstraint.Attachment0 or grabConstraint.Attachment or nil
                local attachment1 = grabConstraint.Attachment1 or nil

                local function belongsToTarget(att)
                    return att and att.Parent and att.Parent:IsDescendantOf(targetChar)
                end

                if belongsToTarget(attachment0) or belongsToTarget(attachment1) then
                    return plr
                end
            end
        end
    end

    return nil
end

local function KnockPlayer(targetPlayer, callback)

    end

        if callback then callback(true) end
        return true
    end

        if callback then callback(false) end
        return false
    end

    KnockActive = true

            KnockActive = false
            knockConnection:Disconnect()
            
            if callback then callback(false) end

    end)
    
    table.insert(ActiveConnections, knockConnection)
    return true

local function ReliableBring(targetPlayer)

        api:notify("Invalid target!", 2)
        return
    end
    
    if not IsKO(targetPlayer) then
        api:notify("Target must be knocked!", 2)
        return
    end
    
    CurrentTarget = targetPlayer
    BringActive = true
    
    local stompOffset = GetValue("ragebot_stomp_offset") or 3.5
    
    api:notify("Bringing " .. targetPlayer.Name, 2)
    
    task.spawn(function()
        local grabbed = false
        local positionLoop
        
        -- Step 1: Continuous positioning until grabbed
        positionLoop = RunService.Heartbeat:Connect(function()
            if grabbed then
                positionLoop:Disconnect()
                return
            end
            
            if not BringActive or not IsKO(targetPlayer) then
                positionLoop:Disconnect()
                BringActive = false
                return
            end
            
            if targetPlayer.Character and targetPlayer.Character:FindFirstChild("UpperTorso") then
                api:set_desync_cframe(CFrame.new(targetPlayer.Character.UpperTorso.Position) + Vector3.new(0, stompOffset, 0))
            end
        end)
        
        -- Step 2: Wait a moment to get in position
        task.wait(0.5)
        
        -- Step 3: PERSISTENT GRAB LOOP (keeps trying until grabbed)
        local grabAttempts = 0
        local maxAttempts = 60
        
        while not grabbed and grabAttempts < maxAttempts and BringActive do
            if not IsKO(targetPlayer) then
                positionLoop:Disconnect()
                BringActive = false
                api:notify("Target got up!", 2)
                return
            end
            
            -- Check if already grabbed
            if targetPlayer.Character:FindFirstChild("GRABBING_CONSTRAINT") then
                grabbed = true
                api:notify("✓ Grabbed " .. targetPlayer.Name, 1)
                break
            end
            
            -- Try to grab
            MainEvent:FireServer("Grabbing")
            grabAttempts = grabAttempts + 1
            
            task.wait(0.2) -- Wait between grab attempts
        end
        
        -- Stop positioning loop
        if positionLoop then
            positionLoop:Disconnect()
        end
        
        -- Check if we actually grabbed them
        if not grabbed or not targetPlayer.Character:FindFirstChild("GRABBING_CONSTRAINT") then
            api:notify("Failed to grab after " .. grabAttempts .. " attempts!", 3)
            BringActive = false
            return
        end
        
        -- Step 4: Stabilize grab
        api:notify("Stabilizing grab...", 1)
        task.wait(0.5)
        
        -- Verify still holding
        if not targetPlayer.Character:FindFirstChild("GRABBING_CONSTRAINT") then
            api:notify("Lost grab!", 2)
            BringActive = false
            return
        end
        
        -- Step 5: Get owner position (ROBUST CHECK)
        local owner = Players:FindFirstChild(OWNER_USERNAME)
        local bringPos = nil
        
        api:notify("Debug: Looking for owner '" .. tostring(OWNER_USERNAME) .. "'", 1)

            if owner.Character then

                     bringPos = owner.Character.HumanoidRootPart.CFrame * CFrame.new(5, 0, 5)
                elseif owner.Character:FindFirstChild("UpperTorso") then
                     bringPos = owner.Character.UpperTorso.CFrame * CFrame.new(5, 0, 5)
                end
            else
                api:notify("Debug: Owner character missing", 2)
            end
        else
            api:notify("Debug: Owner player not found!", 2)
        end
        
        if not bringPos then
            bringPos = CFrame.new(-440, 38, -289) -- Bank default
            api:notify("Debug: Using fallback location (Bank)", 2)
        end
        
        -- Step 6: Teleport Loop (Force Synchronization)
        api:notify("Debug: Teleporting (Looping)...", 1)
        
        -- Loop to force position (Anti-Cheat / Latency bypass)
        for i = 1, 15 do
            if not BringActive then 
                api:notify("Bring cancelled!", 2)
                return 
            end
            
            -- Re-calculate owner position in case they are moving
            local currentOwner = Players:FindFirstChild(OWNER_USERNAME)
            local currentPos = nil
            
            if currentOwner and currentOwner.Character then
                if currentOwner.Character.PrimaryPart then
                    currentPos = currentOwner.Character.PrimaryPart.CFrame * CFrame.new(5, 0, 5)
                elseif currentOwner.Character:FindFirstChild("HumanoidRootPart") then
                    currentPos = currentOwner.Character.HumanoidRootPart.CFrame * CFrame.new(5, 0, 5)

-- BRING ALL function (brings everyone except owner)
local BringAllActive = false
local BringAllList = {}

local function BringAll()
    if BringAllActive then
        api:notify("Bring all already running!", 2)
        return
    end
    
    BringAllActive = true
    BringAllList = {}
    
    -- Get all players except owner and bot
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Name ~= OWNER_USERNAME then
            table.insert(BringAllList, player)
        end
    end
    
    if #BringAllList == 0 then
        api:notify("No players to bring!", 2)
        BringAllActive = false
        return
    end
    
    api:notify("Bringing " .. #BringAllList .. " players...", 3)
    
    task.spawn(function()
        local successCount = 0
        local failCount = 0
        
        for i, targetPlayer in ipairs(BringAllList) do
            if not BringAllActive then
                api:notify("Bring all cancelled!", 2)
                break
            end
            
            api:notify("Bringing " .. i .. "/" .. #BringAllList .. ": " .. targetPlayer.Name, 2)
            
            -- Check if player still exists
            if not targetPlayer or not targetPlayer.Parent then
                failCount = failCount + 1
                continue
            end
            
            if not targetPlayer.Character or not targetPlayer.Character:FindFirstChild("UpperTorso") then
                failCount = failCount + 1
                continue
            end
            
            -- Knock if not knocked
            if not IsKO(targetPlayer) then
                local knockSuccess = false
                
                KnockPlayer(targetPlayer, function(success)
                    knockSuccess = success
                end)
                
                -- Wait for knock to complete
                local waited = 0
                repeat
                    task.wait(0.5)
                    waited = waited + 0.5
                until knockSuccess or waited > 20 or not BringAllActive
                
                if not knockSuccess then
                    api:notify("Failed to knock " .. targetPlayer.Name, 2)
                    failCount = failCount + 1
                    continue
                end
                
                task.wait(0.5)
            end
            
            -- Bring player
            if IsKO(targetPlayer) and BringAllActive then
                local bringSuccess = false
                
                -- Use a callback version
                task.spawn(function()
                    ReliableBring(targetPlayer)
                    bringSuccess = true
                end)
                
                -- Wait for bring to complete
                local waited = 0
                repeat
                    task.wait(0.5)
                    waited = waited + 0.5
                until bringSuccess or waited > 15 or not BringAllActive
                
                if bringSuccess then
                    successCount = successCount + 1
                else
                    failCount = failCount + 1
                end
                
                -- Small delay between brings
                task.wait(2)
            else
                failCount = failCount + 1
            end
        end
        
        api:notify("✓ Bring all complete! Success: " .. successCount .. " | Failed: " .. failCount, 5)
        BringAllActive = false
    end)
end

                    end
                end)
                
                table.insert(ActiveConnections, stompConnection)
            end
        end)
    elseif IsKO(targetPlayer) and not IsDead(targetPlayer) then
        -- Already knocked, just stomp
        CurrentTarget = targetPlayer
        StompActive = true
        
        SetOption("ragebot_targets", targetPlayer.Name)
        api:get_ui_object("ragebot_keybind"):OverrideState(true)
        api:set_ragebot(true)
        
        local stompConnection
        stompConnection = RunService.Heartbeat:Connect(function()
            if IsDead(targetPlayer) then
                -- STOP RAGEBOT
                SetOption("ragebot_targets", {})
                api:get_ui_object("ragebot_keybind"):OverrideState(false)
                api:set_ragebot(false)
                
                StompActive = false
                stompConnection:Disconnect()
            end
        end)
        
        table.insert(ActiveConnections, stompConnection)
    end
end

-- Retaliate against attacker
local function RetaliateAgainst(attacker)
    if not attacker then return end
    
-- Ignore if attacker is owner (owner can’t be “attacked” by bot)
if attacker.Name == OWNER_USERNAME then
    return
end

    
    if RETALIATION_MODE == "loopknock" then
        StartLoopKnock(attacker)
    elseif RETALIATION_MODE == "knock" then
        KnockPlayer(attacker)
    elseif RETALIATION_MODE == "stomp" then
        StompPlayer(attacker, true)
    elseif RETALIATION_MODE == "kill" then
        TargetList[attacker.Name] = true
        UpdateRagebotTargets()
        SetOption("ragebot_use_selected", true)
        api:set_ragebot(true)
    end
end

-- Updated owner protection monitoring (SIMPLIFIED)
local function SetupOwnerProtection()
    task.spawn(function()
        while true do
            task.wait(0.5) -- Check every 0.5 seconds
            
            if not OwnerProtectionEnabled then
                task.wait(2)
                continue
            end
            
            local owner = Players:FindFirstChild(OWNER_USERNAME)
            if not owner or not owner.Character then
                continue
            end
            
            -- Check if auto-save is enabled and owner is knocked
            if AutoSaveEnabled and IsKO(owner) and not SaveInProgress then
                FastAutoSave(owner)
                task.wait(10) -- Cooldown after save attempt
            end
            
            local humanoid = owner.Character:FindFirstChild("Humanoid")
            if not humanoid then
                continue
            end
            
            -- Monitor health changes for retaliation
            local lastHealth = humanoid.Health
            
            local healthConnection
            healthConnection = humanoid.HealthChanged:Connect(function(newHealth)
                if not OwnerProtectionEnabled then
                    healthConnection:Disconnect()
                    return
                end
                
                if newHealth < lastHealth then
                    local bodyEffects = owner.Character:FindFirstChild("BodyEffects")
                    if bodyEffects then
                        local attacker = bodyEffects:FindFirstChild("Attacker")
                        if attacker and attacker.Value then
                            local attackerPlayer = Players:FindFirstChild(attacker.Value)
                            if attackerPlayer and attackerPlayer ~= LocalPlayer then
                                RetaliateAgainst(attackerPlayer)
                            end
                        end
                    end
                end
                
                lastHealth = newHealth
            end)
            
            task.wait(3)
        end
    end)
end

-- FAST & CLEAN Auto Save (NO SPAM, SINGLE GRAB)
local SaveInProgress = false

local function FastAutoSave(owner)
    if not owner or not owner.Character then return end
    if not IsKO(owner) then return end
    if SaveInProgress then return end
    
    SaveInProgress = true
    
    local savePos = SaveLocations[AutoSaveLocation]
    if not savePos then 
        api:notify("No save location!", 2)
        SaveInProgress = false
        return 
    end
    
    if not owner.Character:FindFirstChild("UpperTorso") then 
        SaveInProgress = false
        return 
    end
    
    api:notify("Saving owner...", 1)
    
    task.spawn(function()
        local stompOffset = GetValue("ragebot_stomp_offset") or 3.5
        
        -- Step 1: Get in position quickly
        for i = 1, 3 do
            if owner.Character and owner.Character:FindFirstChild("UpperTorso") then
                local targetPos = owner.Character.UpperTorso.Position + Vector3.new(0, stompOffset, 0)
                api:set_desync_cframe(CFrame.new(targetPos))
            end
            task.wait(0.1)
        end
        
        -- Step 2: Single grab attempt
        if not IsKO(owner) then
            SaveInProgress = false
            return
        end
        
        MainEvent:FireServer("Grabbing")
        
        -- Step 3: Wait for grab to register (check multiple times)
        local grabbed = false
        for i = 1, 15 do
            task.wait(0.1)
            if owner.Character:FindFirstChild("GRABBING_CONSTRAINT") then
                grabbed = true
                break
            end
        end
        
        if not grabbed then
            api:notify("Grab failed - retrying once", 2)
            
            -- ONE retry if first attempt fails
            MainEvent:FireServer("Grabbing")
            task.wait(0.3)
            
            if not owner.Character:FindFirstChild("GRABBING_CONSTRAINT") then
                api:notify("Could not grab owner!", 2)
                SaveInProgress = false
                return
            end
        end
        
        -- Step 4: Stabilize grab
        api:notify("Owner grabbed!", 1)
        task.wait(0.3)
        
        -- Verify still holding
        if not owner.Character:FindFirstChild("GRABBING_CONSTRAINT") then
            api:notify("Lost grab!", 2)
            SaveInProgress = false
            return
        end
        
        -- Step 5: Teleport to save location
        api:teleport(CFrame.new(savePos) + Vector3.new(0, 5, 0))
        task.wait(0.5)
        
        -- Step 6: Drop owner (single drop)
        if owner.Character:FindFirstChild("GRABBING_CONSTRAINT") then
            MainEvent:FireServer("Grabbing")
            api:notify("Dropped at " .. AutoSaveLocation, 1)
        end
        
        task.wait(0.2)
        
        -- Step 7: Void
        VoidBot(2)
        
        api:notify("Owner saved!", 2)
        SaveInProgress = false
    end)
end

-- Guess "carrier" as closest player to a low-HP target
local function GetProbableCarrier(targetPlayer)
    if not targetPlayer or not targetPlayer.Character or not targetPlayer.Character.PrimaryPart then
        return nil
    end

    local targetHealth = GetHealth(targetPlayer)
    -- Only try this if they are low HP
    if targetHealth > 40 then
        return nil
    end

    local targetPos = targetPlayer.Character.PrimaryPart.Position
    local closestPlayer = nil
    local closestDist = math.huge

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= targetPlayer and plr ~= LocalPlayer and plr.Name ~= OWNER_USERNAME and plr.Character and plr.Character.PrimaryPart then
            local dist = (plr.Character.PrimaryPart.Position - targetPos).Magnitude
            if dist < 15 and dist < closestDist then
                closestDist = dist
                closestPlayer = plr
            end
        end
    end

    return closestPlayer
end

        end)

        local waited = 0

            task.wait(0.5)
            waited = waited + 0.5
        until carrierKnocked or waited > 20

        if not carrierKnocked then
            api:notify("Failed to knock " .. carrier.Name, 2)

        end
    end

    -- Normal auto-knock + bring on ORIGINAL target
    if not IsKO(targetPlayer) then
        api:notify("Auto-knocking " .. targetPlayer.Name, 2)
        KnockPlayer(targetPlayer, function(success)
            if success then
                task.wait(0.5)
                ReliableBring(targetPlayer)
            end
        end)

    end

-- Process chat command
local function ProcessCommand(player, message)
    if not string.sub(message, 1, #COMMAND_PREFIX) == COMMAND_PREFIX then
        return
    end
    
    if player.Name ~= OWNER_USERNAME then
        return
    end
    
    local command, args, argString = ParseArgs(message, COMMAND_PREFIX)
    
    if not command then
        return
    end
    
    command = string.lower(command)
    
    -- Check for alias
    if CommandAliases[command] then
        command = CommandAliases[command]
    end
    
    if Commands[command] then
        local success, err = pcall(Commands[command], player, args)
        if not success then
            api:notify("Error: " .. tostring(err), 2)
        end
    end
end

            SetOption("ragebot_use_selected", true)
            api:set_ragebot(true) 
            api:notify("Added & Targeting: " .. target.DisplayName, 2)
        end
        
    elseif cmd == "add_selection" and arg1 then
        -- SELECTOR TOOL COMMAND (No Auto-Rage)
        local target = Get_Player(arg1)
        if target then
            TargetList[target.Name] = true
            UpdateRagebotTargets()
            
            -- Set to use selected targets, but DO NOT enable ragebot
            SetOption("ragebot_use_selected", true)
            
            api:notify("Selected: " .. target.DisplayName .. " (Ready)", 2)
        end
        
    elseif cmd == "remove" and arg1 then
        local target = Get_Player(arg1)
        if target and TargetList[target.Name] then
            TargetList[target.Name] = nil
            UpdateRagebotTargets()
            api:notify("Removed: " .. target.DisplayName, 2)
        end
        
    elseif cmd == "clear" then
        TargetList = {}
        UpdateRagebotTargets()
        api:set_ragebot(false)
        api:notify("Cleared Targets", 2)
        
    elseif cmd == "bring_target" and arg1 then
        local target = Get_Player(arg1)
        if target then
             -- BRING TARGET (Knock -> Bring -> Drop)
             api:notify("Bringing " .. target.Name, 2)
             
             task.spawn(function()
                 if not IsKO(target) then
                     KnockPlayer(target, function(success)
                         if success then
                             task.wait(0.5)
                             ReliableBring(target)
                         end
                     end)
                 else
                     ReliableBring(target)
                 end
             end)
        end
        
    elseif cmd == "knock" and arg1 then
        local target = Get_Player(arg1)
        if target then
            KnockPlayer(target)
        end
        
    elseif cmd == "stomp" and arg1 then
        local target = Get_Player(arg1)
        if target then
            StompPlayer(target, true)
        end
        
    elseif cmd == "rage" and arg1 then
        local enabled = (arg1 == "true")
        api:set_ragebot(enabled)
        
    elseif cmd == "void_prot" and arg1 then
        local enabled = (arg1 == "true")
        SetVoidEnabled(enabled)
        
    elseif cmd == "fake" and arg1 then
        local enabled = (arg1 == "true")
        SetOption("protection_fake_position", enabled)
        api:set_fake(enabled)
        
    elseif cmd == "summon" then
        local owner = Players:FindFirstChild(OWNER_USERNAME)
        if owner and owner.Character and owner.Character.PrimaryPart then
            api:teleport(owner.Character.PrimaryPart.CFrame * CFrame.new(5, 0, 5))
        end
        
    elseif cmd == "stop" then
        StopAllActivity()
        
    else
        -- DYNAMIC COMMAND MAPPING
        -- Check if this command exists in our Chat Commands table
        -- Handle aliases first
        if CommandAliases[cmd] then
            cmd = CommandAliases[cmd]
        end
        
        if Commands[cmd] then
            -- Execute the command handler
            -- We pass 'nil' as the player because these functions might expect it, 
            -- but mostly they rely on 'args'. Key functions using 'player' for checks might need adjustment
            -- but standard commands like .follow, .kill just read args.
            
            local success, err = pcall(Commands[cmd], nil, args)
            if not success then
                api:notify("Cmd Error: " .. tostring(err), 2)
            else
                api:notify("Exec: ." .. cmd, 1)
            end
        end
    end
end

-- File Listener Loop
task.spawn(function()
    while true do
        task.wait(0.3) -- Fast polling
        
        if isfile and isfile(COMM_FILE) then
            local success, content = pcall(readfile, COMM_FILE)
            if success and content ~= "" then
                local jsonSuccess, data = pcall(HttpService.JSONDecode, HttpService, content)
                
                if jsonSuccess and data and data.req_id then
                    -- Check if this is a new command
                    if data.req_id ~= LastReqID then
                        LastReqID = data.req_id
                        ProcessFileCommand(data)
                    end
                end
            end
        end
    end
end)
