-- Levenshtein Distance
local function LevenshteinDistance(s1, s2)
    local len1, len2 = #s1, #s2
    
    if len1 == 0 then return len2 end
    if len2 == 0 then return len1 end
    
    local matrix = {}
    for i = 0, len1 do
        matrix[i] = {[0] = i}
    end
    for j = 0, len2 do
        matrix[0][j] = j
    end
    
    for i = 1, len1 do
        for j = 1, len2 do
            local cost = (s1:sub(i, i) == s2:sub(j, j)) and 0 or 1
            matrix[i][j] = math.min(
                matrix[i-1][j] + 1,
                matrix[i][j-1] + 1,
                matrix[i-1][j-1] + cost
            )
        end
    end
    
    return matrix[len1][len2]
end

-- Get character weapon/item folder
local function GetCharacterItemFolder()
    local char = Players.LocalPlayer.Character
    if not char then return nil end
    return char:FindFirstChild("Items") or char
end

-- Apply visual skin to your character
local function ApplyVisualSkin(category, skinName)
    local char = Players.LocalPlayer.Character
    if not char then
        api:notify("Character not loaded!", 3)
        return false
    end
    
    -- Find weapon matching category
    local itemsFolder = char:FindFirstChild("Items") or char
    local targetItem = nil
    
    for _, item in pairs(itemsFolder:GetChildren()) do
        if item.Name == category:sub(2, -2) then -- Remove brackets
            targetItem = item
            break
        end
    end
    
    if not targetItem then
        api:notify("Weapon not found: " .. category, 3)
        return false
    end
    
    -- Try to find skin configuration
    local skinConfig = targetItem:FindFirstChild("SkinName")
    if skinConfig then
        skinConfig.Value = skinName
        api:notify("Applied visual skin: " .. skinName, 2)
        return true
    end
    
    -- Alternative: look for SkinId
    local skinId = targetItem:FindFirstChild("SkinId")
    if skinId then
        skinId.Value = skinName
        api:notify("Applied visual skin: " .. skinName, 2)
        return true
    end
    
    -- Create new skin value if it doesn't exist
    local newSkin = Instance.new("StringValue")
    newSkin.Name = "SkinName"
    newSkin.Value = skinName
    newSkin.Parent = targetItem
    
    api:notify("Applied visual skin: " .. skinName, 2)
    return true
end

-- Copy player's visual skins
local function CopyPlayerVisuals(targetPlayer)
    if not targetPlayer.Character then
        api:notify("Target has no character!", 3)
        return
    end
    
    local yourChar = Players.LocalPlayer.Character
    if not yourChar then
        api:notify("Your character not loaded!", 3)
        return
    end
    
    local targetItems = targetPlayer.Character:FindFirstChild("Items") or targetPlayer.Character
    local yourItems = yourChar:FindFirstChild("Items") or yourChar
    
    local copiedCount = 0
    
    for _, targetItem in pairs(targetItems:GetChildren()) do
        local skinName = targetItem:FindFirstChild("SkinName")
        if not skinName then
            skinName = targetItem:FindFirstChild("SkinId")
        end
        
        if skinName and skinName.Value ~= "" then
            -- Find matching item in your inventory
            local yourItem = yourItems:FindFirstChild(targetItem.Name)
            if yourItem then
                local yourSkinValue = yourItem:FindFirstChild("SkinName")
                if not yourSkinValue then
                    yourSkinValue = yourItem:FindFirstChild("SkinId")
                end
                
                if yourSkinValue then
                    yourSkinValue.Value = skinName.Value
                    copiedCount = copiedCount + 1
                else
                    local newSkin = Instance.new("StringValue")
                    newSkin.Name = "SkinName"
                    newSkin.Value = skinName.Value
                    newSkin.Parent = yourItem
                    copiedCount = copiedCount + 1
                end
            end
        end
    end
    
    api:notify("✓ Copied " .. copiedCount .. " visual skins from " .. targetPlayer.Name, 3)
end

MainBox:AddButton({
    Text = "Search Skin",
    Func = function()
        local search = Options.skin_search.Value:lower()
        if search == "" then
            api:notify("Type a skin name!", 2)
            return
        end
        
        local skins = SKIN_DATABASE[SELECTED_CATEGORY] or {}
        local found = nil
        
        -- Exact match
        for _, skin in ipairs(skins) do
            if skin:lower() == search then
                found = skin
                break
            end
        end
        
        -- Starts with
        if not found then
            for _, skin in ipairs(skins) do
                if string.sub(skin:lower(), 1, #search) == search then
                    found = skin
                    break
                end
            end
        end
        
        -- Contains
        if not found then
            for _, skin in ipairs(skins) do
                if string.find(skin:lower(), search, 1, true) then
                    found = skin
                    break
                end
            end
        end
        
        if found then
            SELECTED_SKIN = found
            local dropdown = api:get_ui_object("visual_skin")
            dropdown:SetValue(found)
            api:notify("✓ Found: " .. found, 2)
        else
            api:notify("✗ Skin not found", 3)
        end
    end
})
